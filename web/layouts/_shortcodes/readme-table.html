{{- /* A table of the README under a heading, header and cells as written. The README is written in English. A page in another language carries a translation of every row of the table in its front matter, each beside the English it was made from, header first: readmeTables: - heading: Recipes rows: - from: "Key | Where | What it does" to: [Klucz, Gdzie, Co robi] A row of the README with no translation, and a translation of a row the README no longer has, both stop the build, so a change to the README cannot leave a language showing the table as it was. Takes named arguments: heading. */ -}} {{- $heading := .Get "heading" }} {{- $page := .Page }} {{- $rows := partial "readme-table.html" (dict "heading" $heading "header" true) }} {{- if ne site.Language.Lang "en" }} {{- $table := dict }} {{- with where ($page.Params.readmeTables | default slice) "heading" $heading }}{{ $table = index . 0 }}{{ end }} {{- if not $table }} {{- errorf "%s in %s shows the README's table under %q, which is in English, and its front matter has no translation of it under readmeTables" $page.File.Path site.Language.Lang $heading }} {{- end }} {{- $english := slice }} {{- $translated := slice }} {{- range $rows }} {{- $from := delimit .cells " | " }} {{- $english = $english | append $from }} {{- $row := dict }} {{- with where ($table.rows | default slice) "from" $from }}{{ $row = index . 0 }}{{ end }} {{- if not $row }} {{- errorf "%s in %s has no translation of the README's row %q under %q" $page.File.Path site.Language.Lang $from $heading }} {{- else if ne (len $row.to) (len .cells) }} {{- errorf "%s in %s translates the README's row %q into %d cells, and it has %d" $page.File.Path site.Language.Lang $from (len $row.to) (len .cells) }} {{- else }} {{- $translated = $translated | append (dict "cells" $row.to) }} {{- end }} {{- end }} {{- range $table.rows }} {{- if not (in $english .from) }} {{- errorf "%s in %s translates %q, a row the README no longer has under %q" $page.File.Path site.Language.Lang .from $heading }} {{- end }} {{- end }} {{- $rows = $translated }} {{- end }} {{- /* Empty only when a translation is missing, which has stopped the build already. */ -}} {{- with $rows }}
{{ range (index . 0).cells }}{{ end }} {{- range after 1 . }} {{ range .cells }}{{ end }} {{- end }}
